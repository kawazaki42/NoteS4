```haskell
F :: Bool -> Bool -> Bool -> Bool -> Bool
F x1 x2 x3 x4 =
    not x1 && not x3 && not x4 ||
    not x2 &&     x3 ||
        x1 && not x2
```

```hs
a \|/ b = not (a || b)

a \|/ b = not a && not b

a && b = not a \|/ not b

F x1 x2 x3 x4 =
    not (x1 \|/ x3) \|/ x4 ||
    x2 \|/ not x3 ||
    not x1 \|/ x2
```
