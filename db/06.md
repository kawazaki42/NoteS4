```sql
SELECT t.*
FROM r
JOIN (
    SELECT f1, MAX(f2) as m
    FROM r
    GROUP BY f1
    HAVING COUNT(f1) > 1
) AS t1 ON t.f1 = t1.f1 AND t.f2 = t1.m
```
